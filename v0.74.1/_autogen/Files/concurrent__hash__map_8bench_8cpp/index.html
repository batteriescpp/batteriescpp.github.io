// Copyright 2026 Anthony Paul Astolfi
//
#include <batteries/concurrent_hash_map.hpp>
//

#if BATT_CPP_20

#include <benchmark/benchmark.h>

#include <batteries/async/slice_work.hpp>
#include <batteries/int_types.hpp>
#include <batteries/optional.hpp>

#include <random>

namespace {

using namespace batt::int_types;

using batt::None;
using batt::Optional;

constexpr usize kMaxKeys = 5 * 1000 * 1000;
constexpr usize kOpsTotal = 10 * 1000 * 1000;

class ConcurrentHashMapBenchmark : public ::benchmark::Fixture
{
   public:
    void SetUp(::benchmark::State& state) override
    {
        this->query_pct_ = state.range(0);
        this->preload_count_ = kOpsTotal * (100 - this->query_pct_) / 100;
        this->preload_for_thread_.emplace(this->preload_count_, state.threads());
        this->ops_for_thread_.emplace(kOpsTotal, state.threads());

        BATT_CHECK_LE(this->query_pct_, 100);
    }

    void TearDown(::benchmark::State& state [[maybe_unused]]) override
    {
    }

    //+++++++++++-+-+--+----- --- -- -  -  -   -

    Optional<batt::DivideWork> preload_for_thread_;
    Optional<batt::DivideWork> ops_for_thread_;
    usize preload_count_ = 0;
    i32 query_pct_ = 0;
    batt::ConcurrentHashMap<u64, u64> hash_map_;
};

BENCHMARK_DEFINE_F(ConcurrentHashMapBenchmark, Scalability)(benchmark::State& state)
{
    std::default_random_engine rng{/*seed=*/(u32)state.thread_index()};
    std::uniform_int_distribution<u64> pick_key{0, kMaxKeys};
    std::uniform_int_distribution<i32> pick_pct{0, 99};

    // Pre-benchmark setup.
    //
    {
        const usize preload_this_thread = (*this->preload_for_thread_)(state.thread_index());

        for (usize i = 0; i < preload_this_thread; ++i) {
            const u64 key = pick_key(rng);
            this->hash_map_.upsert(key, i);
        }
    }

    const usize ops_this_thread = (*this->ops_for_thread_)(state.thread_index());

    // Benchmark.
    //
    for (auto _ : state) {
        u64 result_total = 0;

        for (usize i = 0; i < ops_this_thread; ++i) {
            const u64 key = pick_key(rng);

            if (pick_pct(rng) < this->query_pct_) {
                //----- --- -- -  -  -   -
                // Query
                //
                if (this->hash_map_.lookup(key)) {
                    ++result_total;
                }
            } else {
                //----- --- -- -  -  -   -
                // Upsert
                //
                if (this->hash_map_.upsert(key, i)) {
                    ++result_total;
                }
            }
        }

        BATT_CHECK_NE(result_total, ~((u64)state.thread_index()));
    }

    // Post-benchmark cleanup.
}

BENCHMARK_REGISTER_F(ConcurrentHashMapBenchmark, Scalability)
    ->ThreadRange(1, std::thread::hardware_concurrency())

    // Make sure we have enough samples to smooth out variation.
    //
    ->MinTime(2.0 /*seconds*/)

    // Run for different percentages of upserts vs queries.  Parameter is:
    //   `{preload_count, pct_queries}`
    //
    ->Args({0})    // no queries
    ->Args({5})    // 5% queries
    ->Args({25})   // 25% queries
    ->Args({50})   // 50% queries
    ->Args({75})   // 75% queries
    ->Args({95})   // 95% queries
    ->Args({100})  // 95% queries
    ;

}  //namespace

#endif  // BATT_CPP_20
