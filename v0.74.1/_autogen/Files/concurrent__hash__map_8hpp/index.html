struct Erasure {
    static constexpr bool kAlwaysMatches = true;

    //+++++++++++-+-+--+----- --- -- -  -  -   -

    template <typename Value>
    bool matches(Value&&) const noexcept
    {
        return true;
    }
};

//=#=#==#==#===============+=+=+=+=++=++++++++++++++-++-+--+-+----+---------------
//
template <typename PredicateFn>
struct ConditionalErasure {
    static constexpr bool kAlwaysMatches = false;

    //+++++++++++-+-+--+----- --- -- -  -  -   -

    PredicateFn&& predicate;

    template <typename Value>
    bool matches(Value&& value) const noexcept
    {
        return this->predicate(BATT_FORWARD(value));
    }
};

struct IsErasure<ConditionalErasure<PredicateFn>> : std::true_type {};

struct Constants {
    static constexpr i32 kDefaultSizeLog2 = 8;

    static constexpr usize kDefaultMaxProbeDepth = 16;

    static constexpr usize kMinMigrationChunkSize = 512;

    static constexpr Fingerprint32 kBucketHasValueMask = 1;

    static constexpr Fingerprint32 kBucketInChainMask = 2;

    static constexpr Fingerprint32 kBucketTentativeMask = 4;

    static constexpr Fingerprint32 kBucketMigratedMask = 8;

    static constexpr i32 kBucketFingerprintShift = 4;
};

//=#=#==#==#===============+=+=+=+=++=++++++++++++++-++-+--+-+----+---------------
//
struct BucketBase : Constants {
    using Self = BucketBase;

    //+++++++++++-+-+--+----- --- -- -  -  -   -

    static constexpr Fingerprint32 fingerprint_from_hash_value(u64 hash_value) noexcept
    {
        return ((hash_value >> 32) & 0xfffffff0ul) | (Self::kBucketHasValueMask | Self::kBucketInChainMask);
    }

    static constexpr bool has_value(Fingerprint32 observed_fingerprint) noexcept
    {
        return (observed_fingerprint & kBucketHasValueMask) != 0;
    }

    static constexpr bool in_chain(Fingerprint32 observed_fingerprint) noexcept
    {
        return (observed_fingerprint & kBucketInChainMask) != 0;
    }

    static constexpr bool is_tentative(Fingerprint32 observed_fingerprint) noexcept
    {
        return (observed_fingerprint & kBucketTentativeMask) != 0;
    }

    static constexpr bool is_migrated(Fingerprint32 observed_fingerprint) noexcept
    {
        return (observed_fingerprint & kBucketMigratedMask) != 0;
    }

    //+++++++++++-+-+--+----- --- -- -  -  -   -

    BucketMutex mutex_;

    std::atomic<Fingerprint32> fingerprint_ = 0;

    //+++++++++++-+-+--+----- --- -- -  -  -   -

    BucketBase(const Self&) = delete;

    Self& operator=(const Self&) = delete;

    //+++++++++++-+-+--+----- --- -- -  -  -   -

    bool has_value() const noexcept
    {
        return Self::has_value(this->fingerprint_.load());
    }

    bool in_chain() const noexcept
    {
        return Self::in_chain(this->fingerprint_.load());
    }

    bool is_migrated() const noexcept
    {
        return Self::is_migrated(this->fingerprint_.load());
    }
};

static_assert(sizeof(BucketBase) == 8);

struct BucketArrayBase : Constants {
    using Self = BucketArrayBase;

    //+++++++++++-+-+--+----- --- -- -  -  -   -

    const usize size_;

    usize index_mask_;

    std::atomic<usize> max_probe_depth_;

    const i64 generation_;

    const i32 size_log2_;

    //+++++++++++-+-+--+----- --- -- -  -  -   -

    explicit BucketArrayBase(i32 size_log2, i64 generation) noexcept
        : size_{usize{1} << size_log2}
        , index_mask_{this->size_ - 1}
        , max_probe_depth_{Self::kDefaultMaxProbeDepth}
        , generation_{generation}
        , size_log2_{size_log2}
    {
    }

    BucketArrayBase(const Self&) = delete;

    BucketArrayBase& operator=(const Self&) = delete;

    //+++++++++++-+-+--+----- --- -- -  -  -   -

    usize size() const noexcept
    {
        return usize{1} << this->size_log2_;
    }
};

struct WorkProgress {
    std::atomic<usize> next_index_ = 0;

    std::atomic<usize> done_count_ = 0;

    //+++++++++++-+-+--+----- --- -- -  -  -   -

    WorkProgress() = default;

    WorkProgress(const WorkProgress& other) noexcept
        : next_index_{other.next_index_.load()}
        , done_count_{other.done_count_.load()}
    {
    }

    //+++++++++++-+-+--+----- --- -- -  -  -   -

    void reset() noexcept
    {
        this->next_index_.store(0);
        this->done_count_.store(0);
    }

    WorkProgress& operator=(const WorkProgress& other) noexcept
    {
        if (this != &other) {
            this->next_index_.store(other.next_index_.load());
            this->done_count_.store(other.done_count_.load());
        }
        return *this;
    }
};

//
template <typename Key, typename Value, typename HashFn = std::hash<Key>, typename EqFn = std::equal_to<Key>>
    requires(std::is_trivial_v<Key> /*&& std::is_trivial_v<Value>*/)
class ConcurrentHashMap : public concurrent_hash_map::Constants
{
    /*
      Invariants:

      A bucket's migrated bit is never unset
      A bucket's in_chain bit is never unset
      A bucket is never updated after its migrated bit is set
      A bucket's tentative bit only goes from 0 -> 1 together with the has value bit;
         once the has value bit has been observed without the tentative bit, the tentative bit is never set
         again unless has value is cleared first.
      has-value alweays implies in-chain
     */

   public:
    using Self = ConcurrentHashMap;

    using KeyStorage = std::aligned_storage_t<sizeof(Key), alignof(Key)>;

    using ValueStorage = std::aligned_storage_t<sizeof(Value), alignof(Value)>;

    //+++++++++++-+-+--+----- --- -- -  -  -   -

    using Fingerprint32 = concurrent_hash_map::Fingerprint32;
    using HashValue64 = concurrent_hash_map::HashValue64;

    //+++++++++++-+-+--+----- --- -- -  -  -   -

    using LookupResult = concurrent_hash_map::LookupResult;
    using UpdateResult = concurrent_hash_map::UpdateResult;
    using InsertResult = concurrent_hash_map::InsertResult;
    using UpsertResult = concurrent_hash_map::UpsertResult;

    using Erasure = concurrent_hash_map::Erasure;

    template <typename PredicateFn>
    using ConditionalErasure = concurrent_hash_map::ConditionalErasure<PredicateFn>;

    using LazyInit = concurrent_hash_map::LazyInit;

    using BucketMutex = concurrent_hash_map::BucketMutex;

    using SearchControl = concurrent_hash_map::SearchControl;

    using WorkProgress = concurrent_hash_map::WorkProgress;

    //+++++++++++-+-+--+----- --- -- -  -  -   -
    //
    struct Bucket : concurrent_hash_map::BucketBase {
        using Self = Bucket;
        using Super = concurrent_hash_map::BucketBase;

        //----- --- -- -  -  -   -

        KeyStorage key_storage_;

        ValueStorage value_storage_;

        //----- --- -- -  -  -   -

        using Super::Super;

        //----- --- -- -  -  -   -

        const Key& key() noexcept
        {
            return *reinterpret_cast<Key*>(&this->key_storage_);
        }

        Value& value() noexcept
        {
            return *reinterpret_cast<Value*>(&this->value_storage_);
        }

        template <typename KeyArg, typename... ValueArgs>
        void emplace(const BucketMutex::WriteLock&, KeyArg&& key_arg, ValueArgs&&... value_args) noexcept
        {
            BATT_ASSERT(!this->has_value());

            new (&this->key_storage_) Key{BATT_FORWARD(key_arg)};
            new (&this->value_storage_) Value{BATT_FORWARD(value_args)...};
        }
    };

    struct BucketArray;

    template <typename KeyArg>
    struct KeySearch;

    //+++++++++++-+-+--+----- --- -- -  -  -   -
    //
    struct BucketArray : concurrent_hash_map::BucketArrayBase {
        using Self = BucketArray;
        using Super = concurrent_hash_map::BucketArrayBase;

        //----- --- -- -  -  -   -

        using AlignedBucket = CpuCacheLineIsolated<Bucket>;

        using BucketStorage = std::aligned_storage_t<sizeof(AlignedBucket), alignof(AlignedBucket)>;

        //----- --- -- -  -  -   -

        std::unique_ptr<BucketStorage[]> buckets_;

        //----- --- -- -  -  -   -

        template <typename... InitMode>
        explicit BucketArray(i32 size_log2, i64 generation, InitMode...) noexcept
            : Super{size_log2, generation}
            , buckets_{new BucketStorage[this->size_]}
        {
            if constexpr (!std::is_same_v<std::tuple<std::decay_t<InitMode>...>, std::tuple<LazyInit>>) {
                this->initialize();
            }
        }

        //----- --- -- -  -  -   -

        void initialize() noexcept
        {
            this->initialize(0, this->size());
        }

        void initialize(usize start, usize count) noexcept
        {
            void* const addr = &this->buckets_[start];
            usize const byte_count = sizeof(BucketStorage) * count;

            std::memset(addr, 0, byte_count);
        }

        Bucket& operator[](usize index) noexcept
        {
            BATT_ASSERT_LT(index, this->size());
            return **reinterpret_cast<AlignedBucket*>(&this->buckets_[index]);
        }

        Bucket& operator()(u64 hash_value) noexcept
        {
            return this->operator[](hash_value & this->index_mask_);
        }

        template <typename KeyArg, typename ValueArg>
        LookupResult lookup(u64 hash_value, const EqFn& key_equals, const KeyArg& key,
                            ValueArg& value_out) noexcept;

        template <typename KeyArg, typename ValueArg>
        UpdateResult update(const KeySearch<KeyArg>& search, const ValueArg& value_arg) noexcept;

        template <typename KeyArg, typename ValueArg>
        InsertResult insert(const KeySearch<KeyArg>& search, const ValueArg& value_arg) noexcept;

        template <typename KeyArg, typename ValueArg>
        UpsertResult upsert(u64 hash_value, const EqFn& key_equals, const KeyArg& key_arg,
                            const ValueArg& value_arg) noexcept;

        void migrate(const HashFn& hash_fn, const EqFn& key_equals, usize start, usize count,
                     BucketArray* target) noexcept;

        usize count() noexcept
        {
            usize total = 0;
            for (usize i = 0; i < this->size(); ++i) {
                Bucket& bucket = this->operator[](i);
                const u64 observed_fingerprint = bucket.fingerprint_.load();
                if (Bucket::has_value(observed_fingerprint) && !Bucket::is_tentative(observed_fingerprint)) {
                    ++total;
                }
            }
            return total;
        }
    };

    //+++++++++++-+-+--+----- --- -- -  -  -   -
    //
    struct State {
        std::shared_ptr<BucketArray> current_;
    };

    struct Migration {
        std::shared_ptr<BucketArray> current_;

        std::shared_ptr<BucketArray> future_;

        WorkProgress initialize_progress_;

        WorkProgress migrate_progress_;
    };

    //+++++++++++-+-+--+----- --- -- -  -  -   -

    ConcurrentHashMap() noexcept : ConcurrentHashMap{/*size_hint=*/usize{1} << kDefaultSizeLog2}
    {
    }

    explicit ConcurrentHashMap(usize size_hint) noexcept
        : state_{}
        , key_hash_{}
        , key_equals_{}
        , migration_mutex_{}
        , migration_{nullptr}
        , next_generation_{2}
    {
        ToggleWriter<State> state_writer{this->state_};

        state_writer.new_value().current_ =
            std::make_shared<BucketArray>(batt::log2_ceil(size_hint), /*generation=*/1);

        BATT_CHECK_EQ(this->next_generation_, state_writer.new_value().current_->generation_ + 1);
    }

    //+++++++++++-+-+--+----- --- -- -  -  -   -

    usize size() noexcept
    {
        ToggleReader<State> state_reader{this->state_};
        return state_reader->current_->count();
    }

    template <typename KeyArg, typename ValueArg>
    bool lookup(const KeyArg& key, ValueArg& value_out) noexcept
    {
        const u64 hash_value = this->key_hash_(key);

        for (;;) {
            const LookupResult result = [&] {
                ToggleReader<State> state_reader{this->state_};
                return state_reader->current_->lookup(hash_value, this->key_equals_, key, value_out);
            }();

            switch (result) {
            case LookupResult::kNotFound:
                return false;

            case LookupResult::kFound:
                return true;

            case LookupResult::kMigrated:
                this->help_with_migration();
                break;

            default:
                BATT_PANIC() << "Bad LookupResult value: " << (i32)result;
                BATT_UNREACHABLE();
            }
        }
    }

    template <typename KeyArg>
    Optional<Value> lookup(const KeyArg& key) noexcept
    {
        Value value;
        if (!this->lookup(key, value)) {
            return None;
        }
        return value;
    }

    template <typename KeyArg, typename ValueArg>
    bool upsert(const KeyArg& key, ValueArg&& value) noexcept
    {
        const u64 hash_value = this->key_hash_(key);

        for (;;) {
            std::shared_ptr<BucketArray> current = nullptr;

            const UpsertResult result = [&] {
                ToggleReader<State> state_reader{this->state_};

                const UpsertResult result =
                    state_reader->current_->upsert(hash_value, this->key_equals_, key, BATT_FORWARD(value));

                if (result == UpsertResult::kFull) {
                    current = state_reader->current_;
                }
                return result;
            }();

            switch (result) {
            case UpsertResult::kUpdated:
                return false;

            case UpsertResult::kInserted:
                return true;

            case UpsertResult::kMigrated:
                this->help_with_migration();
                break;

            case UpsertResult::kFull:
                this->start_migration(std::move(current));
                break;

            default:
                BATT_PANIC() << "Bad UpsertResult value: " << (i32)result;
                BATT_UNREACHABLE();
            }
        }
    }

    template <typename KeyArg, typename ValueArg>
    bool insert(const KeyArg& key, ValueArg&& value) noexcept
    {
        const u64 hash_value = this->key_hash_(key);

        for (;;) {
            std::shared_ptr<BucketArray> current = nullptr;

            const InsertResult result = [&] {
                ToggleReader<State> state_reader{this->state_};
                KeySearch<KeyArg> key_search{*state_reader->current_, hash_value, this->key_equals_, key};

                const InsertResult result = state_reader->current_->insert(key_search, value);

                if (result == InsertResult::kFull) {
                    current = state_reader->current_;
                }
                return result;
            }();

            switch (result) {
            case InsertResult::kInserted:
                return true;

            case InsertResult::kConflict:
                return false;

            case InsertResult::kMigrated:
                this->help_with_migration();
                break;

            case InsertResult::kFull:
                this->start_migration(std::move(current));
                break;

            default:
                BATT_PANIC() << "Bad LookupResult value: " << (i32)result;
                BATT_UNREACHABLE();
            }
        }
    }

    template <typename KeyArg, typename ValueArg>
    bool update(const KeyArg& key, ValueArg&& value) noexcept
    {
        const u64 hash_value = this->key_hash_(key);

        for (;;) {
            const UpdateResult result = [&] {
                ToggleReader<State> state_reader{this->state_};
                KeySearch<KeyArg> key_search{*state_reader->current_, hash_value, this->key_equals_, key};

                return state_reader->current_->update(key_search, value);
            }();

            switch (result) {
            case UpdateResult::kNotFound:
                return false;

            case UpdateResult::kUpdated:
                return true;

            case UpdateResult::kMigrated:
                this->help_with_migration();
                break;

            default:
                BATT_PANIC() << "Bad LookupResult value: " << (i32)result;
                BATT_UNREACHABLE();
            }
        }
    }

    template <typename KeyArg>
    bool erase(const KeyArg& key) noexcept
    {
        return this->update(key, Erasure{});
    }

    template <typename KeyArg, std::invocable<const Value&> PredicateFn>
    bool erase_if(const KeyArg& key, PredicateFn&& predicate) noexcept
    {
        return this->update(key, ConditionalErasure<PredicateFn>{BATT_FORWARD(predicate)});
    }

    //+++++++++++-+-+--+----- --- -- -  -  -   -
   private:
    static usize migration_chunk_size(usize total_size) noexcept
    {
        return std::max<usize>(kMinMigrationChunkSize, (total_size / std::thread::hardware_concurrency()));
    }

    //+++++++++++-+-+--+----- --- -- -  -  -   -

    void start_migration(std::shared_ptr<BucketArray> current) noexcept
    {
        const i32 next_size_log2 = current->size_log2_ + 1;
        const i64 next_generation = current->generation_ + 1;

        std::shared_ptr<Migration> migration = nullptr;
        {
            std::scoped_lock<std::mutex> lock{this->migration_mutex_};

            if (!this->migration_) {
                // In the unlikely event that some other thread has started a migration for `current` *and*
                // that migration has already completed, just return (and retry the original operation).
                //
                if (this->next_generation_ != next_generation) {
                    return;
                }

                this->migration_ = std::make_shared<Migration>();
                this->migration_->current_ = current;
                this->migration_->future_ =
                    std::make_shared<BucketArray>(next_size_log2, next_generation, LazyInit{});
            }
            migration = this->migration_;
        }
        this->help_with_migration(std::move(migration));
    }

    void help_with_migration() noexcept
    {
        std::shared_ptr<Migration> migration = [this] {
            std::scoped_lock<std::mutex> lock{this->migration_mutex_};
            return this->migration_;
        }();
        if (!migration) {
            return;
        }
        this->help_with_migration(std::move(migration));
    }

    void help_with_migration(std::shared_ptr<Migration> migration) noexcept
    {
        //+++++++++++-+-+--+----- --- -- -  -  -   -
        // Phase 1: Initialize the future bucket array in parallel on all threads.
        //
        const bool last_initialization_thread = this->process_work_slices(  //
            migration->future_->size(), migration->initialize_progress_,
            [&migration](usize index, usize count) {
                migration->future_->initialize(index, count);
            });

        //----- --- -- -  -  -   -
        // Wait until all initialization chunks are completed, or notify if this was the last thread.
        //
        if (last_initialization_thread) {
            this->finish_initialization(migration);
        } else {
            this->wait_for_initialization(migration);
        }

        //+++++++++++-+-+--+----- --- -- -  -  -   -
        // Phase 2: Insert all keys from the current array into future_.
        //
        const bool last_migration_thread = this->process_work_slices(  //
            migration->current_->size(), migration->migrate_progress_, [&](usize index, usize count) {
                migration->current_->migrate(this->key_hash_, this->key_equals_, index, count,
                                             migration_->future_.get());
            });

        //----- --- -- -  -  -   -
        // Wait until the migration completes, or finalize the migration if this was the last thread.
        //
        if (last_migration_thread) {
            this->finish_migration();
        } else {
            this->wait_for_migration(std::move(migration));
        }
    }

    template <std::invocable<usize, usize> WorkFn>
    bool process_work_slices(usize total_size, WorkProgress& progress, WorkFn&& work_fn) noexcept
    {
        const usize target_chunk_size = Self::migration_chunk_size(total_size);
        bool last_one_out = false;

        for (;;) {
            const usize prior_value = progress.next_index_.fetch_add(target_chunk_size);
            if (prior_value >= total_size) {
                break;
            }

            const usize chunk_start = prior_value;
            const usize chunk_size = std::min(target_chunk_size, total_size - chunk_start);

            work_fn(chunk_start, chunk_size);

            const usize prior_done = progress.done_count_.fetch_add(chunk_size);
            if (prior_done + chunk_size == total_size) {
                last_one_out = true;
                break;
            }
            BATT_CHECK_LT(prior_done + chunk_size, total_size);
        }

        return last_one_out;
    }

    void finish_initialization(const std::shared_ptr<Migration>& migration) noexcept
    {
        migration->initialize_progress_.done_count_.notify_all();
    }

    void finish_migration() noexcept
    {
        {
            std::scoped_lock<std::mutex> lock{this->migration_mutex_};
            BATT_CHECK_NOT_NULLPTR(this->migration_);

            // Activate the new BucketArray.
            //
            {
                ToggleWriter<State> state_writer{this->state_};
                BATT_CHECK_EQ(state_writer.old_value().current_, this->migration_->current_);
                state_writer.new_value().current_ = this->migration_->future_;
            }

            // Reset the migration state and advance the next generation counter.
            //
            this->migration_ = nullptr;
            ++this->next_generation_;
        }
        this->state_.sync();
    }

    void wait_for_initialization(const std::shared_ptr<Migration>& migration) noexcept
    {
        for (;;) {
            const usize observed_done = migration->initialize_progress_.done_count_.load();
            if (observed_done >= migration->future_->size()) {
                break;
            }
            migration->initialize_progress_.done_count_.wait(observed_done);
        }
    }

    void wait_for_migration(const std::shared_ptr<Migration>& migration) noexcept
    {
        for (;;) {
            u64 seq = 0;
            {
                ToggleReader<State> state_reader{this->state_};
                if (state_reader->current_->generation_ - migration->current_->generation_ > 0) {
                    return;
                }
                seq = state_reader.seq();
            }
            this->state_.wait(seq);
        }
    }

    //+++++++++++-+-+--+----- --- -- -  -  -   -

    Toggle<State> state_;

    HashFn key_hash_;

    EqFn key_equals_;

    std::mutex migration_mutex_;

    std::shared_ptr<Migration> migration_;

    i64 next_generation_;
};

//=#=#==#==#===============+=+=+=+=++=++++++++++++++-++-+--+-+----+---------------

template <typename Key, typename Value, typename HashFn, typename EqFn>
    requires(std::is_trivial_v<Key> /*&& std::is_trivial_v<Value>*/)
template <typename KeyArg>
struct ConcurrentHashMap<Key, Value, HashFn, EqFn>::KeySearch {
    BucketArray& bucket_array_;
    u64 hash_value_;
    const EqFn& key_equals_;
    const KeyArg& key_arg_;

    u64 fingerprint_pattern_;
    u64 probe_begin_;
    u64 probe_end_;

    //----- --- -- -  -  -   -

    explicit KeySearch(BucketArray& bucket_array, u64 hash_value, const EqFn& key_equals,
                       const KeyArg& key_arg) noexcept
        : bucket_array_{bucket_array}
        , hash_value_{hash_value}
        , key_equals_{key_equals}
        , key_arg_{key_arg}
        , fingerprint_pattern_{Bucket::fingerprint_from_hash_value(this->hash_value_)}
        , probe_begin_{hash_value}
        , probe_end_{this->probe_begin_ + this->bucket_array_.max_probe_depth_.load()}
    {
    }

    //----- --- -- -  -  -   -

    template <std::invocable<const KeySearch&, BucketMutex::ReadLock&, Bucket&, u64 /*probe_index*/> Fn>
    SearchControl operator()(Fn&& fn) const noexcept
    {
        for (u64 probe_index = this->probe_begin_; probe_index != this->probe_end_;) {
            Bucket& bucket = this->bucket_array_(probe_index);
            const u64 observed_fingerprint = bucket.fingerprint_.load();

            // Check for a match.
            //
            if (observed_fingerprint == this->fingerprint_pattern_) {
                // Observe the seq mutex value.
                //
                BucketMutex::ReadLock read_lock{bucket.mutex_};

                // Double-check the fingerprint after acquiring the read lock.
                //
                const u64 observed_after_read_lock = bucket.fingerprint_.load();
                if (observed_after_read_lock != this->fingerprint_pattern_) {
                    continue;
                }

                // Optimistically read the key.
                //
                const Key bucket_key = bucket.key();

                // Make sure we got a clean read on the key; otherwise retry.
                //
                if (read_lock.changed()) {
                    continue;
                }

                // Key read is clean; do full comparison.
                //
                if (this->key_equals_(bucket_key, this->key_arg_)) {
                    // We have a match!  Read the value.
                    //
                    const SearchControl action = fn(*this, read_lock, bucket, probe_index);

                    if (action == SearchControl::kReturnMigrated) {
                        return SearchControl::kReturnMigrated;
                    }

                    if (action == SearchControl::kRetry) {
                        probe_index = this->probe_begin_;
                        continue;
                    }

                    BATT_ASSERT(action == SearchControl::kDoneFound ||
                                action == SearchControl::kDoneNotFound);

                    return action;
                }
            }

            // If the bucket's migrated bit is set, then bail out and help with the migration before
            // retrying.
            //
            if (Bucket::is_migrated(observed_fingerprint)) {
                return SearchControl::kReturnMigrated;
            }

            // If we have probed beyond the end of the current chain, then stop (not found).
            //
            if (!Bucket::in_chain(observed_fingerprint)) {
                break;
            }

            // Keep probing...
            //
            ++probe_index;
        }
        // (probe loop)

        return SearchControl::kDoneNotFound;
    }
};

//==#==========+==+=+=++=+++++++++++-+-+--+----- --- -- -  -  -   -
//
template <typename Key, typename Value, typename HashFn, typename EqFn>
    requires(std::is_trivial_v<Key> /*&& std::is_trivial_v<Value>*/)
template <typename KeyArg, typename ValueArg>
auto ConcurrentHashMap<Key, Value, HashFn, EqFn>::BucketArray::lookup(u64 hash_value, const EqFn& key_equals,
                                                                      const KeyArg& key,
                                                                      ValueArg& value_out) noexcept
    -> LookupResult
{
    KeySearch<KeyArg> search{*this, hash_value, key_equals, key};

    const SearchControl search_result =
        search([&value_out](const KeySearch<KeyArg>&, BucketMutex::ReadLock& read_lock, Bucket& bucket,
                            u64 /*probe_index*/) {
            // We have a match!  Read the value.
            //
            value_out = bucket.value();

            // If we didn't get a clean read on the value, then retry from the start.
            //
            if (read_lock.changed()) {
                return SearchControl::kRetry;
            }

            return SearchControl::kDoneFound;
        });

    BATT_ASSERT_NE(search_result, SearchControl::kRetry);

    return static_cast<LookupResult>(search_result);
}

//==#==========+==+=+=++=+++++++++++-+-+--+----- --- -- -  -  -   -
//
template <typename Key, typename Value, typename HashFn, typename EqFn>
    requires(std::is_trivial_v<Key> /*&& std::is_trivial_v<Value>*/)
template <typename KeyArg, typename ValueArg>
auto ConcurrentHashMap<Key, Value, HashFn, EqFn>::BucketArray::update(const KeySearch<KeyArg>& search,
                                                                      const ValueArg& value_arg) noexcept
    -> UpdateResult
{
    constexpr bool kIsErase = concurrent_hash_map::IsErasure<std::decay_t<ValueArg>>::value;

    const SearchControl search_result = search(                                                   //
        [&value_arg](const KeySearch<KeyArg>&, BucketMutex::ReadLock& read_lock, Bucket& bucket,  //
                     u64 /*probe_index*/)                                                         //
        {
            if constexpr (kIsErase) {
#if !BATT_COMPILER_IS_MSVC
                //
                // The Microsoft compiler has a bug where it tries to evaluate the following conditional
                // expression, even when kIsErase == false.

                if constexpr (!std::decay_t<ValueArg>::kAlwaysMatches)
#endif  // BATT_COMPILER_IS_MSVC
                {
                    Value observed_value = bucket.value();
                    if (read_lock.changed()) {
                        return SearchControl::kRetry;
                    }
                    if (!value_arg.matches(observed_value)) {
                        return SearchControl::kDoneNotFound;
                    }
                }
            }

            // The key was found!  Acquire a write lock and attempt to update; note, this may fail
            // if another thread gets in first and deletes the key.
            //
            BucketMutex::WriteLock write_lock{bucket.mutex_};
            const u64 observed_fingerprint_after_lock = bucket.fingerprint_.load();

            if (Bucket::is_migrated(observed_fingerprint_after_lock)) {
                return SearchControl::kReturnMigrated;
            }

            if (!Bucket::has_value(observed_fingerprint_after_lock) || write_lock.changed_since(read_lock)) {
                return SearchControl::kRetry;
            }

            if constexpr (kIsErase) {
                if (!value_arg.matches(bucket.value())) {
                    return SearchControl::kDoneNotFound;
                }
                bucket.fingerprint_.fetch_and(~Self::kBucketHasValueMask);
            } else {
                bucket.value() = value_arg;
            }

            return SearchControl::kDoneFound;
        });

    BATT_ASSERT_NE(search_result, SearchControl::kRetry);

    // The enum values of LookupResult and UpdateResult are aligned, so just cast and return.
    //
    return static_cast<UpdateResult>(search_result);
}

//==#==========+==+=+=++=+++++++++++-+-+--+----- --- -- -  -  -   -
//
template <typename Key, typename Value, typename HashFn, typename EqFn>
    requires(std::is_trivial_v<Key> /*&& std::is_trivial_v<Value>*/)
template <typename KeyArg, typename ValueArg>
auto ConcurrentHashMap<Key, Value, HashFn, EqFn>::BucketArray::insert(const KeySearch<KeyArg>& search,
                                                                      const ValueArg& value_arg) noexcept
    -> InsertResult
{
    // Iterate over the probe range to find a free bucket.
    //
    for (u64 probe_index = search.probe_begin_; probe_index != search.probe_end_;) {
        Bucket& bucket = this->operator()(probe_index);
        const u64 observed_fingerprint = bucket.fingerprint_.load();

        // If a migration is detected, stop and help out.
        //
        if (Bucket::is_migrated(observed_fingerprint)) {
            return InsertResult::kMigrated;
        }

        // If the bucket isn't empty, then keep probing...
        //
        if (Bucket::has_value(observed_fingerprint)) {
            ++probe_index;
            continue;
        }

        // Found a bucket without a value; acquire an exclusive lock and do a tentative insert.
        //
        {
            BucketMutex::WriteLock write_lock{bucket.mutex_};
            const u64 observed_after_write_lock = bucket.fingerprint_.load();

            // Double-check the fingerprint after acquiring the write lock.
            //
            if (observed_after_write_lock != observed_fingerprint) {
                continue;
            }

            // Insert the key/value pair.
            //
            bucket.emplace(write_lock, search.key_arg_, value_arg);

            // Update the bucket's fingerprint to claim it, adding the tentative bit; the insertion will
            // not be visible to readers (including updaters) until we clear this bit.
            //
            bucket.fingerprint_.store(search.fingerprint_pattern_ | Self::kBucketTentativeMask);
        }

        // Most paths out of the loop main loop require the insertion to be reverted, so use batt::finally
        // and just cancel on the one success path.
        //
        auto revert_bucket = batt::finally([&] {
            bucket.fingerprint_.fetch_and(~(Self::kBucketHasValueMask | Self::kBucketTentativeMask));
        });

        // Now that we have claimed a bucket, scan though all the possible buckets for this key, checking
        // for delete/insert conflicts.
        //
        for (usize check_index = search.probe_begin_; check_index < search.probe_end_;) {
            if (check_index == probe_index) {
                ++check_index;
                continue;
            }
            // ^^^
            // Only check buckets _other than_ the one we just claimed...

            Bucket& check_bucket = this->operator()(check_index);
            const u64 checked_fingerprint = check_bucket.fingerprint_.load();

            // A bucket with the same key is a conflict whether tentative or not, so mask out the tentative
            // bit when comparing.
            //
            if ((checked_fingerprint & ~Self::kBucketTentativeMask) == search.fingerprint_pattern_) {
                BucketMutex::ReadLock read_lock{check_bucket.mutex_};

                // Double-check the fingerprint after acquiring the read lock.
                //
                const u64 checked_after_read_lock = check_bucket.fingerprint_.load();
                if (checked_after_read_lock != checked_fingerprint) {
                    continue;
                }

                // Optimistically read the key.
                //
                const Key bucket_key = check_bucket.key();

                // Make sure we got a clean read on the key; otherwise revert and retry.
                //
                if (read_lock.changed()) {
                    continue;
                }

                // The key read was clean; do a full comparison.
                //
                if (search.key_equals_(bucket_key, search.key_arg_)) {
                    // If there is a conflict, then we back off and retry.  Because it is possible for
                    // multiple threads to both claim a bucket, we vary the number of yields according to
                    // our position in the probe chain, to make repeat collisions less likely (this is
                    // roughly analogous to random jitter in backoff timers).
                    //
                    {
                        // Important: we want to revert *first*, so that if another thread backs off with
                        // fewer yields, it will not see our tentative insertion again.
                        //
                        auto revert_bucket_now = std::move(revert_bucket);
                    }
                    for (usize backoff = search.probe_begin_; backoff < probe_index; ++backoff) {
                        Task::yield();
                    }
                    return InsertResult::kConflict;
                    //
                    // ~batt::finally reverts our claim on the tentative bucket
                }

                // No match; move on to the next bucket.
                //
                ++check_index;
                continue;
            }
            // (if possible match)

            if (Bucket::is_migrated(checked_fingerprint)) {
                return InsertResult::kMigrated;
                //
                // ~batt::finally reverts our claim on the tentative bucket
            }

            // If we step off the end of the chain, then stop here; any concurrent insertions for the same
            // key which end up extending the chain *must* see our insertion (possibly tentative), so we
            // are good.
            //
            if (!Bucket::in_chain(checked_fingerprint)) {
                break;
            }

            // This bucket does not conflict; keep going...
            //
            ++check_index;
        }
        // (conflict check loop)

        // Success!  Clear the tentative bit to finalize the insertion.
        //
        revert_bucket.cancel();
        bucket.fingerprint_.fetch_and(~Self::kBucketTentativeMask);
        return InsertResult::kInserted;
    }
    // (probe loop)

    // If we get to the end of the probe chain without finding a free slot, then we may need to kick off a
    // migration.  It is best for a caller to try at least one more time to do an update in case, since
    // migrations are so expensive.
    //
    return InsertResult::kFull;
}

//==#==========+==+=+=++=+++++++++++-+-+--+----- --- -- -  -  -   -
//
template <typename Key, typename Value, typename HashFn, typename EqFn>
    requires(std::is_trivial_v<Key> /*&& std::is_trivial_v<Value>*/)
template <typename KeyArg, typename ValueArg>
auto ConcurrentHashMap<Key, Value, HashFn, EqFn>::BucketArray::upsert(u64 hash_value, const EqFn& key_equals,
                                                                      const KeyArg& key_arg,
                                                                      const ValueArg& value_arg) noexcept
    -> UpsertResult
{
    // BucketArray::upsert handles inserts, updates, and erasures; erasure of a key is specified by
    // passing ConcurrentHashMap::Erasure{} as the value_arg.
    //
    constexpr bool kIsErase = concurrent_hash_map::IsErasure<std::decay_t<ValueArg>>::value;

    KeySearch<KeyArg> search{*this, hash_value, key_equals, key_arg};

    bool retrying_update_once_more_after_full = false;

    // Loop until we get a decisive result.
    //
    for (;;) {
        // First assume the key is already present and attempt to update.
        //
        const UpdateResult update_result = this->update(search, value_arg);

        // If update already failed once because we failed to allocate a bucket, then stop here.
        //
        if (retrying_update_once_more_after_full) {
            return UpsertResult::kFull;
        }

        // Only fall-through to the insert case if this is *not* a delete and update failed to find the key.
        //
        if (kIsErase || update_result != UpdateResult::kNotFound) {
            return static_cast<UpsertResult>(update_result);
        }

        if constexpr (!kIsErase) {
            const InsertResult insert_result = this->insert(search, value_arg);

            // Retry on conflict; `insert` has already backed off, so continue immediately.
            //
            if (insert_result == InsertResult::kConflict) {
                continue;
            }

            // If we fail to allocate a free bucket, then attempt to update one more time before failing.
            //
            if (insert_result == InsertResult::kFull) {
                BATT_CHECK(!retrying_update_once_more_after_full);
                retrying_update_once_more_after_full = true;
                continue;
            }

            // The insert_result was either kInserted or kMigrated; return in both cases.
            //
            return static_cast<UpsertResult>(insert_result);
        }
    }
}

//==#==========+==+=+=++=+++++++++++-+-+--+----- --- -- -  -  -   -
//
template <typename Key, typename Value, typename HashFn, typename EqFn>
    requires(std::is_trivial_v<Key> /*&& std::is_trivial_v<Value>*/)
void ConcurrentHashMap<Key, Value, HashFn, EqFn>::BucketArray::migrate(const HashFn& hash_fn,
                                                                       const EqFn& key_equals, usize index,
                                                                       usize count,
                                                                       BucketArray* target) noexcept
{
    const usize last = index + count;
    BATT_CHECK_LE(last, this->size());

    for (; index < last; ++index) {
        Bucket& bucket = this->operator[](index);

        // We must acquire a write lock on the bucket because we are going to mark it as migrated.
        //
        BucketMutex::WriteLock write_lock{bucket.mutex_};

        u64 observed_fingerprint;
        for (;;) {
            observed_fingerprint = bucket.fingerprint_.load();
            if (!Bucket::is_tentative(observed_fingerprint)) {
                break;
            }
            Task::yield();
        }

        BATT_CHECK(!Bucket::is_migrated(observed_fingerprint))
            << BATT_INSPECT(this->generation_) << BATT_INSPECT(observed_fingerprint) << BATT_INSPECT(index);

        if (Bucket::has_value(observed_fingerprint)) {
            const u64 hash_value = hash_fn(bucket.key());
            for (;;) {
                const usize old_max_probe_depth = target->max_probe_depth_.load();

                const UpsertResult result =
                    target->upsert(hash_value, key_equals, bucket.key(), std::move(bucket.value()));

                BATT_CHECK_NE(result, UpsertResult::kMigrated);
                BATT_CHECK_NE(result, UpsertResult::kUpdated);

                if (result == UpsertResult::kInserted) {
                    break;
                }

                BATT_CHECK_EQ(result, UpsertResult::kFull);
                BATT_CHECK_LT(old_max_probe_depth, target->size());

                // Increase the probe depth and retry.
                //
                atomic_clamp_min(target->max_probe_depth_, old_max_probe_depth + kDefaultMaxProbeDepth);
            }
        }

        // Mark the bucket as migrate (whether or not it contained a value).
        //
        bucket.fingerprint_.fetch_or(Self::kBucketMigratedMask);
    }
}
